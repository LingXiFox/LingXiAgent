#include <iostream>
#include <cmath>

int main() {
    int n = 10000;
    std::cout << "Primes between 0 and 10000:" << std::endl;
    for(int i = 2; i <= n; i++) {
        bool is_prime = true;
        for(int j = 2; j * j <= i; j++) {
            if(i % j == 0) {
                is_prime = false;
                break;
            }
        }
        if(is_prime) {
            std::cout << i << " ";
        }
    }
    std::cout << std::endl;
    return 0;
}
